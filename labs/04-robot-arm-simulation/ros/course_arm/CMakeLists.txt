cmake_minimum_required(VERSION 3.0.2)
project(course_arm)
find_package(catkin REQUIRED COMPONENTS rospy std_msgs sensor_msgs)
catkin_package(CATKIN_DEPENDS rospy std_msgs sensor_msgs)
catkin_install_python(PROGRAMS scripts/pose.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
foreach(folder launch urdf rviz worlds config meshes aiml)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${folder})
    install(DIRECTORY ${folder}/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/${folder})
  endif()
endforeach()
